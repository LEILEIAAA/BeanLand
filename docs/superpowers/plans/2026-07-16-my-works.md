# 我的拼豆（作品墙）实现计划

> **面向 AI 代理的工作者：** 本仓库禁用子代理；在当前会话按任务顺序实现。步骤用复选框（`- [ ]`）跟踪。规格：`docs/superpowers/specs/2026-07-16-my-works-design.md`。

**目标：** 会员页独立「我的拼豆」卡 + 作品墙页（上传 / 列表 / 预览 / 删除），云集合 `works` 持久化；与集章完全独立。

**架构：** 纯函数 `worksView.js` 负责 caption 规范化与会员卡预览映射（TDD）；三个云函数 `listMyWorks` / `createWork` / `deleteWork` 读写 `works` 并删云文件；小程序 `api.js` 封装调用；会员页拉 `limit:3` 预览，作品页全量网格。

**技术栈：** 微信小程序、云开发（数据库 + 云存储 + 云函数）、Node test runner（`node --test`）

```js
const CAPTION_MAX = 40;

function normalizeCaption(caption) {
  const text = String(caption == null ? '' : caption).trim();
  return text.length > CAPTION_MAX ? text.slice(0, CAPTION_MAX) : text;
}

function buildWorksCardPreview(list) {
  const rows = Array.isArray(list) ? list : [];
  const sorted = rows.slice().sort((a, b) => Number(b.createdAt || 0) - Number(a.createdAt || 0));
  const previewThumbs = sorted.slice(0, 3).map((row) => ({
    id: row._id,
    imageFileID: row.imageFileID,
  }));
  const workCount = rows.length;
  const worksEmpty = workCount === 0;
  return {
    workCount,
    previewThumbs,
    worksEmpty,
    worksHint: worksEmpty ? '还没有作品，去拍一张' : '把成品照留下来',
  };
}

function formatWorkDate(ms) {
  const n = Number(ms);
  if (!n) return '';
  const d = new Date(n);
  const y = d.getUTCFullYear();
  const m = String(d.getUTCMonth() + 1).padStart(2, '0');
  const day = String(d.getUTCDate()).padStart(2, '0');
  return `${y}-${m}-${day}`;
}

module.exports = {
  CAPTION_MAX,
  normalizeCaption,
  buildWorksCardPreview,
  formatWorkDate,
};
```

```js
const cloud = require('wx-server-sdk');

cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV });
const db = cloud.database();

exports.main = async (event) => {
  const { OPENID } = cloud.getWXContext();
  if (!OPENID) {
    return { ok: false, code: 'NO_OPENID', message: 'missing openid', list: [], total: 0 };
  }

  const limit = Math.min(Math.max(Number((event && event.limit) || 50), 1), 100);
  const result = await db
    .collection('works')
    .where({ _openid: OPENID })
    .orderBy('createdAt', 'desc')
    .limit(limit)
    .get();

  const list = result.data || [];
  return { ok: true, list, total: list.length };
};
```

说明：预览只需前 3 条时 `total` 为返回条数而非全库 count；会员卡用 `list.length` 作缩略即可。若要准确总数，另加 `count()`——首版 YAGNI，会员卡显示「本批条数」时改为：列表页用 limit 50，卡片用 limit 3 时右侧数字用列表页回写或二次 count。  
**首版约定：** 卡片数字 = `listMyWorks({ limit: 3 })` 返回的 `total` 不够准时，增加可选 `countOnly` 或在 list 时用 `db.collection('works').where({_openid}).count()`。实现时采用：

```js
const countRes = await db.collection('works').where({ _openid: OPENID }).count();
const total = (countRes && countRes.total) || 0;
```

```js
const cloud = require('wx-server-sdk');

cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV });
const db = cloud.database();

const CAPTION_MAX = 40;

function normalizeCaption(caption) {
  const text = String(caption == null ? '' : caption).trim();
  return text.length > CAPTION_MAX ? text.slice(0, CAPTION_MAX) : text;
}

exports.main = async (event) => {
  const { OPENID } = cloud.getWXContext();
  if (!OPENID) {
    return { ok: false, code: 'NO_OPENID', message: 'missing openid' };
  }

  const imageFileID = String((event && event.imageFileID) || '').trim();
  if (!imageFileID.startsWith('cloud://')) {
    return { ok: false, code: 'INVALID_FILE', message: 'invalid imageFileID' };
  }

  const now = Date.now();
  const doc = {
    _openid: OPENID,
    imageFileID,
    caption: normalizeCaption(event && event.caption),
    createdAt: now,
    updatedAt: now,
  };

  const addRes = await db.collection('works').add({ data: doc });
  return { ok: true, work: { _id: addRes._id, ...doc } };
};
```

- [ ] **步骤 4：`deleteWork/index.js`**

```js
const cloud = require('wx-server-sdk');

cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV });
const db = cloud.database();

exports.main = async (event) => {
  const { OPENID } = cloud.getWXContext();
  if (!OPENID) {
    return { ok: false, code: 'NO_OPENID', message: 'missing openid' };
  }

  const workId = String((event && event.workId) || '').trim();
  if (!workId) {
    return { ok: false, code: 'INVALID_ID', message: 'missing workId' };
  }

  const col = db.collection('works');
  const got = await col.doc(workId).get();
  const row = got && got.data;
  if (!row || row._openid !== OPENID) {
    return { ok: false, code: 'NOT_FOUND', message: 'work not found' };
  }

  await col.doc(workId).remove();
  if (row.imageFileID) {
    try {
      await cloud.deleteFile({ fileList: [row.imageFileID] });
    } catch (error) {
      console.warn('[deleteWork] deleteFile failed', error);
    }
  }
  return { ok: true, workId };
};
```

- [ ] **步骤 3：`index.js` 核心逻辑**

```js
const api = require('../../../utils/api');
const { formatWorkDate, normalizeCaption } = require('../../../utils/worksView');

Page({
  data: {
    works: [],
    loading: true,
  },
  onShow() {
    this.load();
  },
  async load() {
    this.setData({ loading: true });
    try {
      const res = await api.listMyWorks({ limit: 50 });
      const list = (res && res.ok && Array.isArray(res.list) ? res.list : []).map((row) => ({
        ...row,
        dateLabel: formatWorkDate(row.createdAt),
      }));
      this.setData({ works: list });
    } catch (error) {
      console.warn('[works] load failed', error);
      wx.showToast({ title: '加载失败', icon: 'none' });
      this.setData({ works: [] });
    } finally {
      this.setData({ loading: false });
    }
  },
  onPreview(event) {
    const file = event.currentTarget.dataset.file;
    const urls = this.data.works.map((w) => w.imageFileID).filter(Boolean);
    wx.previewImage({ current: file, urls });
  },
  async onUpload() {
    try {
      const choose = await wx.chooseMedia({
        count: 1,
        mediaType: ['image'],
        sourceType: ['album', 'camera'],
      });
      const file = choose.tempFiles && choose.tempFiles[0];
      if (!file || !file.tempFilePath) return;

      wx.showLoading({ title: '上传中', mask: true });
      const cloudPath = `works/${Date.now()}-${Math.random().toString(36).slice(2, 8)}.jpg`;
      const up = await wx.cloud.uploadFile({
        cloudPath,
        filePath: file.tempFilePath,
      });
      const caption = normalizeCaption('');
      const res = await api.createWork({
        imageFileID: up.fileID,
        caption,
      });
      wx.hideLoading();
      if (!res || !res.ok) {
        wx.showToast({ title: (res && res.message) || '保存失败', icon: 'none' });
        return;
      }
      wx.showToast({ title: '已保存', icon: 'success' });
      this.load();
    } catch (error) {
      wx.hideLoading();
      if (error && error.errMsg && /cancel/i.test(error.errMsg)) return;
      console.warn('[works] upload failed', error);
      wx.showToast({ title: '上传失败', icon: 'none' });
    }
  },
  onDelete(event) {
    const workId = event.currentTarget.dataset.id;
    wx.showModal({
      title: '删除作品',
      content: '删除后不可恢复',
      success: async (modal) => {
        if (!modal.confirm) return;
        try {
          const res = await api.deleteWork({ workId });
          if (!res || !res.ok) {
            wx.showToast({ title: (res && res.message) || '删除失败', icon: 'none' });
            return;
          }
          wx.showToast({ title: '已删除', icon: 'success' });
          this.load();
        } catch (error) {
          console.warn('[works] delete failed', error);
          wx.showToast({ title: '删除失败', icon: 'none' });
        }
      },
    });
  },
});
```

首版备注可先空串；若要弹输入框，在 `createWork` 前加 `wx.showModal` editable（基础库支持）或跳过备注。YAGNI：首版无备注输入亦可，规格允许 caption 为空。
